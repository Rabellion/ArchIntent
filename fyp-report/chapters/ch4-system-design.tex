% ENTITY LIST READ FROM archintent.sql -- 34 tables, 38 foreign keys.
% Framework tables (migrations, cache, cache_locks, jobs, job_batches,
% failed_jobs, personal_access_tokens) are infrastructure, not domain
% entities, and are deliberately NOT listed below. Neither are the four
% tables suffixed "_old", which are superseded.
%
% ACTION REQUIRED: your schema still contains portfolios_old,
% portfolio_images_old, contractor_portfolios_old and
% contractor_portfolio_images_old. Drop them before you export the
% schema for your ER diagram, or an examiner will ask why there are two
% portfolio designs in the same database.

The data model comprises the following domain entities.

\begin{itemize}[leftmargin=*, itemsep=1pt]
  \item \textbf{users} -- the single account table for all four roles.
        Primary key \texttt{user\_id}; holds \texttt{full\_name},
        \texttt{email}, \texttt{password\_hash}, a \texttt{role}
        enumeration of client, architect, contractor and admin,
        \texttt{account\_status}, identity document fields, and a
        \texttt{deleted\_at} timestamp for soft deletion.
  \item \textbf{architects} and \textbf{contractors} -- role-specific
        professional profiles, each extending a \texttt{users} row.
  \item \textbf{architect\_portfolios} and \textbf{architect\_projects}
        (with \textbf{architect\_project\_images}) -- an architect's
        portfolio and the individual works within it, including the
        style tags used by the matching engine.
  \item \textbf{contractor\_portfolios}, \textbf{contractor\_projects}
        and \textbf{contractor\_project\_images} -- the equivalent
        structures for contractors.
  \item \textbf{projects} -- a client's project, referencing the client
        and, once chosen, a \texttt{selected\_architect\_id} and
        \texttt{selected\_contractor\_id}.
  \item \textbf{project\_matches} -- the persisted output of the
        semantic matching engine, linking a project to an architect and
        a specific portfolio project with a match score.
  \item \textbf{design\_revisions} -- revision requests raised against a
        delivered design.
  \item \textbf{bids} -- contractor bids against a posted construction
        job.
  \item \textbf{agreements} and \textbf{agreement\_signatures} -- the
        digital agreement for a project and the signature of each party
        to it.
  \item \textbf{payments} -- payment records, referencing a
        \texttt{payer\_id} and a \texttt{payee\_id}.
  \item \textbf{budz\_wallets}, \textbf{budz\_packages} and
        \textbf{budz\_transactions} -- the platform credit system.
  \item \textbf{conversations} and \textbf{messages} -- the messaging
        subsystem; a conversation links
        \texttt{participant\_one\_id} and
        \texttt{participant\_two\_id}.
  \item \textbf{reviews} -- ratings, referencing a
        \texttt{reviewer\_id} and a \texttt{reviewee\_id}.
  \item \textbf{admin\_logs} -- the administrative audit trail.
\end{itemize}
